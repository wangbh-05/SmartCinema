\section{核心技术设计}

\subsection{空间热度可视化}

\begin{frame}{空间热度可视化：从离散座位到连续热力场}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{问题}
        单个座位的红黄蓝状态，难以看出\textbf{整片区域}的热度趋势。
      \end{block}
      \begin{itemize}
        \setlength{\itemsep}{7pt}
        \item 弧形座标 + 过道间隔，还原影厅空间结构
        \item ``座位层 + 热度层''双 Canvas 叠加
        \item 周一到周日使用确定性扰动：\textbf{同一条件下可复现}
      \end{itemize}
    \column{0.52\textwidth}
      \centering
      \begin{tikzpicture}[font=\small, node distance=4mm]
        \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
          minimum width=38mm, minimum height=9mm] (seat) {座位基础热度：位置 + 已售情况};
        \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
          below=of seat, minimum width=38mm, minimum height=9mm] (day) {星期因子 + 中心区提升};
        \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
          below=of day, minimum width=38mm, minimum height=9mm] (gauss) {高斯核加权插值};
        \node[draw=purple!80!black, fill=purple!20, rounded corners, align=center,
          below=of gauss, minimum width=38mm, minimum height=9mm] (canvas) {连续热力场 + 座位图};
        \draw[->, thick, purple!80!black] (seat) -- (day);
        \draw[->, thick, purple!80!black] (day) -- (gauss);
        \draw[->, thick, purple!80!black] (gauss) -- (canvas);
      \end{tikzpicture}
      \vspace{3mm}
      \scriptsize 热度服务于决策，不伪装成真实商业客流数据。
  \end{columns}
\end{frame}

\subsection{跨端选座交互}

\begin{frame}{跨端选座交互：同一座位图，两种交互语言}
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
      \begin{block}{PC：精确操作}
        \begin{itemize}
          \setlength{\itemsep}{5pt}
          \item 点击单座
          \item 鼠标拖拽框选
          \item 滚轮/按钮缩放
        \end{itemize}
      \end{block}
    \column{0.31\textwidth}
      \begin{block}{手机：手势优先}
        \begin{itemize}
          \setlength{\itemsep}{5pt}
          \item 点按或路径滑选
          \item 单指平移
          \item 双指锚点缩放
        \end{itemize}
      \end{block}
    \column{0.35\textwidth}
      \begin{block}{共同的安全边界}
        \begin{itemize}
          \setlength{\itemsep}{5pt}
          \item 阈值区分点击与拖动
          \item 边界阻尼 + 松手回弹
          \item 高分屏座标换算
        \end{itemize}
      \end{block}
  \end{columns}
  \vspace{3mm}
  \begin{alertblock}{设计原则}
    先识别用户\textbf{意图}，再执行选座、平移或缩放；避免手机上“滑一下就误选座”。
  \end{alertblock}
  \scriptsize 同时保留键盘方向键、虚拟焦点与隐藏语义按钮层，使 Canvas 不成为无障碍黑箱。
\end{frame}

\subsection{智能选座引擎}

\begin{frame}{智能选座引擎：约束优先的可解释推荐}
  \centering
  \begin{tikzpicture}[font=\small, node distance=5mm]
    \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
      minimum width=41mm, minimum height=10mm] (generate) {生成候选：同排 / 相邻排};
    \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
      right=of generate, minimum width=41mm, minimum height=10mm] (filter) {硬约束过滤：库存、无障碍};
    \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
      below=of generate, minimum width=41mm, minimum height=10mm] (rank) {多目标排序：年龄、偏好、价格};
    \node[draw=purple!80!black, fill=purple!20, rounded corners, align=center,
      right=of rank, minimum width=41mm, minimum height=10mm] (output) {输出至多 5 组：方案 + 理由};
    \draw[->, thick, purple!80!black] (generate) -- (filter);
    \draw[->, thick, purple!80!black] (filter) -- (output);
    \draw[->, thick, purple!80!black] (generate) -- (rank);
    \draw[->, thick, purple!80!black] (rank) -- (output);
  \end{tikzpicture}
  \vspace{5mm}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{不可突破的规则}
        无台阶通行、陪同席配对、不可用座位、票数一致、禁止制造孤座。
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{可说明的取舍}
        儿童避开前三排、长者避开后三排；资源不足时分级放宽，并明确告诉用户原因。
      \end{block}
  \end{columns}
\end{frame}

\subsection{AI 偏好理解}

\begin{frame}{AI 偏好理解：规则负责最终决策}
  \begin{columns}[T,onlytextwidth]
    \column{0.42\textwidth}
      \centering
      \begin{tikzpicture}[font=\small, node distance=4mm]
        \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
          minimum width=35mm, minimum height=9mm] (text) {一句自然语言偏好};
        \node[draw=purple!80!black, fill=purple!10, rounded corners, align=center,
          below=of text, minimum width=35mm, minimum height=9mm] (intent) {结构化意图\\目标 + 权重 + 约束};
        \node[draw=purple!80!black, fill=purple!20, rounded corners, align=center,
          below=of intent, minimum width=35mm, minimum height=9mm] (engine) {本地合法候选排序\\座位 + 解释};
        \draw[->, thick, purple!80!black] (text) -- (intent);
        \draw[->, thick, purple!80!black] (intent) -- (engine);
      \end{tikzpicture}
    \column{0.55\textwidth}
      \begin{block}{6 个比较维度}
        \small 银幕距离、居中程度、周边空位、价格水平、少受打扰、进出便捷。
      \end{block}
      \begin{itemize}\small
        \setlength{\itemsep}{3pt}
        \item DeepSeek 仅负责理解文本
        \item 8 秒超时或失败时规则引擎兜底
        \item 库存与无障碍由本地规则裁决
      \end{itemize}
  \end{columns}
  \vspace{2mm}
  \begin{alertblock}{核心边界}
    \small AI 不生成座位号；它只提供可审计的偏好，本地算法只在\textbf{合法候选集}内排序。
  \end{alertblock}
\end{frame}

\subsection{AI 协作与团队开发流程}

\begin{frame}{AI 协作开发流程}
  % 说明：工具名称与使用细节请按团队实际情况核对调整
  \begin{itemize}
    \setlength{\itemsep}{4pt}
    \item \textbf{使用工具}：基于 VS Code 使用 AI 编程助手（Claude Code / GitHub Copilot 等）辅助开发，覆盖需求拆解、代码生成、代码审查与测试编写
    \item \textbf{AI 生成内容}：Canvas 绘制逻辑、推荐算法初版、单元测试用例、说明文档初稿等
    \item \textbf{学生修改与决策}：对 AI 生成代码逐行审查、修改与验证；产品功能取舍、推荐规则、交互与视觉设计均由成员独立决策
    \item \textbf{质量把关}：AI 输出必须通过单元测试与代码评审后方可合入，与团队 Git 协作流程保持一致
    \item \textbf{可追溯性}：编写 AI 开发说明书，记录产品设计、用户分析、AI 生成内容、学生修改思路与设计决策；参考资料与大模型使用均准确标注
  \end{itemize}
\end{frame}

\begin{frame}{团队协作开发流程}
  \begin{itemize}
    \setlength{\itemsep}{4pt}
    \item \textbf{版本管理与分支策略}：基于 GitHub 仓库协作，\texttt{main} 维护稳定版本、\texttt{develop} 作为开发集成分支、成员在各自功能分支上并行开发，明确区分稳定版与开发版
    \item \textbf{合并前质量保障}：单元测试（座位数据 / 推荐算法 / 评分引擎）+ Pull Request 代码评审；整体回归测试与优化通过后再合入稳定分支
    \item \textbf{例会与进度管理}：每周例会汇报个人进展、问题与后续计划；借助需求清单 / 里程碑逐项跟踪与验收
    \item \textbf{文档与规范}：统一提交信息规范，维护 README、实现说明等文档，保证开发过程可追溯
  \end{itemize}
\end{frame}

\subsection{心得体会}

\begin{frame}{心得体会}
  \begin{itemize}
    \setlength{\itemsep}{4pt}
    \item \textbf{产品思维}：围绕用户画像（情侣 / 家庭 / 老人 / 团体）设计功能，把“选座决策”从用户转移到系统，真正落实“别让用户思考”
    \item \textbf{前端工程}：Canvas 手写绘制、模块分层、CSS 变量与响应式、无障碍规范，体会到“可维护”比“能跑”更重要
    \item \textbf{AI 协作}：AI 是效率加速器，但正确性与产品判断必须由自己负责，AI 生成内容要审查、要验证
    \item \textbf{团队协作}：Git 分支并行 + 评审 + 例会机制，让多人开发既高效又可追溯
    \item \textbf{交付与规范}：文档、测试与提交规范同样重要，是项目按时落地的保障
  \end{itemize}
\end{frame}
